# AI v3.7 EXE Installer Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build a reproducible Windows GUI installer named `SU_AI_PNG_Export_v3.7_Setup.exe` for the Illustrator CEP v3.7 plugin.

**Architecture:** A .NET Framework 4.8 WinForms executable embeds the plugin as a ZIP resource named `PluginPayload`. A separate installer core owns safe extraction, exact-directory replacement, rollback, CEP registry setup, and uninstall; the form only presents status and invokes that core. A PowerShell build script creates the payload, compiles tests and the installer with the Windows-provided C# compiler, verifies metadata, and copies the final EXE to the repository root.

**Tech Stack:** C# 5 / .NET Framework 4.8, WinForms, `System.IO.Compression`, Windows Registry, PowerShell 7, Windows `csc.exe`, Git.

## Global Constraints

- Output filename: `SU_AI_PNG_Export_v3.7_Setup.exe`.
- File and product version: `3.7.0.0`; product description: `SU+AI PNG Export Installer`; company: `段土土`.
- Install only to `%APPDATA%\Adobe\CEP\extensions\su-ai-png-export` by default and never require administrator privileges.
- Only replace or remove the exact `su-ai-png-export` directory; never touch sibling Adobe extensions, export files, Illustrator documents, or the SU plugin.
- Set `PlayerDebugMode=1` as a string under current-user `CSXS.9` through `CSXS.20`; leave those shared values in place on uninstall.
- Keep the Illustrator panel and plugin business code unchanged.
- Use only Windows/.NET components already present on the machine; do not add package-manager or installer-framework dependencies.
- The output remains unsigned, matching the v3.6 installer behavior.

- [ ] **Step 1: Write the failing core test executable**

Create `InstallerCoreTests.cs` with a minimal assertion runner. The test reads the `PluginPayload` resource and source directory argument, installs into a unique temporary extensions root, then checks exact file hashes, safe repeat installation, sibling preservation, invalid-payload protection, and uninstall scope:

```csharp
using System;
using System.Collections.Generic;
using System.IO;
using System.IO.Compression;
using System.Reflection;
using System.Security.Cryptography;

namespace SUAIInstaller {
    internal static class InstallerCoreTests {
        private static int assertions;

        private static void Assert(bool condition, string message) {
            assertions++;
            if (!condition) throw new Exception(message);
        }

        private static string Hash(string path) {
            using (var sha = SHA256.Create())
            using (var input = File.OpenRead(path)) {
                return BitConverter.ToString(sha.ComputeHash(input)).Replace("-", "");
            }
        }

        private static Dictionary<string, string> FileHashes(string root) {
            var result = new Dictionary<string, string>(StringComparer.OrdinalIgnoreCase);
            foreach (var path in Directory.GetFiles(root, "*", SearchOption.AllDirectories)) {
                result[path.Substring(root.Length).TrimStart(Path.DirectorySeparatorChar)] = Hash(path);
            }
            return result;
        }

        private static Stream Payload() {
            return Assembly.GetExecutingAssembly().GetManifestResourceStream("PluginPayload");
        }

        public static int Main(string[] args) {
            var source = Path.GetFullPath(args[0]);
            var temp = Path.Combine(Path.GetTempPath(), "suai-installer-test-" + Guid.NewGuid().ToString("N"));
            var extensions = Path.Combine(temp, "extensions");
            var sibling = Path.Combine(extensions, "keep-me");
            Directory.CreateDirectory(sibling);
            File.WriteAllText(Path.Combine(sibling, "marker.txt"), "keep");
            try {
                using (var payload = Payload()) InstallerCore.Install(payload, extensions, false);
                var target = InstallerCore.TargetPath(extensions);
                var expected = FileHashes(source);
                var actual = FileHashes(target);
                Assert(expected.Count == 12, "Expected 12 source files");
                Assert(actual.Count == expected.Count, "Installed file count differs");
                foreach (var pair in expected) Assert(actual.ContainsKey(pair.Key) && actual[pair.Key] == pair.Value, "Installed file mismatch: " + pair.Key);

                File.WriteAllText(Path.Combine(target, "obsolete.txt"), "old");
                using (var payload = Payload()) InstallerCore.Install(payload, extensions, false);
                Assert(!File.Exists(Path.Combine(target, "obsolete.txt")), "Repeat install left obsolete file");

                var marker = Path.Combine(target, "existing.txt");
                File.WriteAllText(marker, "existing");
                using (var invalid = new MemoryStream()) {
                    using (var zip = new ZipArchive(invalid, ZipArchiveMode.Create, true)) {
                        var entry = zip.CreateEntry("bad.txt");
                        using (var writer = new StreamWriter(entry.Open())) writer.Write("bad");
                    }
                    invalid.Position = 0;
                    try { InstallerCore.Install(invalid, extensions, false); Assert(false, "Invalid payload was accepted"); }
                    catch (InvalidDataException) { }
                }
                Assert(File.ReadAllText(marker) == "existing", "Invalid payload changed existing install");

                InstallerCore.Uninstall(extensions);
                Assert(!Directory.Exists(target), "Target remained after uninstall");
                Assert(File.ReadAllText(Path.Combine(sibling, "marker.txt")) == "keep", "Sibling extension was changed");
                Console.WriteLine("PASS InstallerCore: " + assertions + " assertions");
                return 0;
            } catch (Exception error) {
                Console.Error.WriteLine("FAIL InstallerCore: " + error);
                return 1;
            } finally {
                if (Directory.Exists(temp)) Directory.Delete(temp, true);
            }
        }
    }
}
```

- [ ] **Step 3: Implement path safety, ZIP validation, install rollback, registry setup, and exact uninstall**

Create `InstallerCore.cs` around this interface and control flow:

```csharp
using Microsoft.Win32;
using System;
using System.IO;
using System.IO.Compression;

namespace SUAIInstaller {
    internal static class InstallerCore {
        internal const string ExtensionName = "su-ai-png-export";

        internal static string DefaultExtensionsRoot() {
            return Path.Combine(Environment.GetFolderPath(Environment.SpecialFolder.ApplicationData), "Adobe", "CEP", "extensions");
        }

        internal static string TargetPath(string extensionsRoot) {
            if (String.IsNullOrWhiteSpace(extensionsRoot)) throw new ArgumentException("Extensions root is required.", "extensionsRoot");
            return Path.Combine(Path.GetFullPath(extensionsRoot), ExtensionName);
        }

        internal static void ValidatePayload(Stream payload) {
            if (payload == null) throw new InvalidDataException("Missing PluginPayload resource.");
            using (var copy = CopyToMemory(payload))
            using (var zip = new ZipArchive(copy, ZipArchiveMode.Read, true)) {
                bool manifest = false, host = false;
                foreach (var entry in zip.Entries) {
                    var name = entry.FullName.Replace('\\', '/');
                    if (String.Equals(name, "CSXS/manifest.xml", StringComparison.OrdinalIgnoreCase)) manifest = true;
                    if (String.Equals(name, "host/main.jsx", StringComparison.OrdinalIgnoreCase)) host = true;
                }
                if (!manifest || !host) throw new InvalidDataException("PluginPayload is missing required v3.7 files.");
            }
        }

        internal static void Install(Stream payload, string extensionsRoot, bool enableDebugMode) {
            using (var bytes = CopyToMemory(payload)) {
                ValidatePayload(bytes);
                bytes.Position = 0;
                var root = Path.GetFullPath(extensionsRoot);
                var target = TargetPath(root);
                var staging = Path.Combine(root, "." + ExtensionName + ".staging-" + Guid.NewGuid().ToString("N"));
                var backup = Path.Combine(root, "." + ExtensionName + ".backup-" + Guid.NewGuid().ToString("N"));
                Directory.CreateDirectory(root);
                bool movedOld = false, movedNew = false;
                try {
                    ExtractSafely(bytes, staging);
                    ValidateExtracted(staging);
                    if (Directory.Exists(target)) { Directory.Move(target, backup); movedOld = true; }
                    Directory.Move(staging, target); movedNew = true;
                    if (enableDebugMode) EnableCepDebugMode();
                    if (Directory.Exists(backup)) Directory.Delete(backup, true);
                } catch {
                    Exception restoreError = null;
                    try { if (movedNew && Directory.Exists(target)) Directory.Delete(target, true); }
                    catch (Exception error) { restoreError = error; }
                    try { if (movedOld && Directory.Exists(backup) && !Directory.Exists(target)) Directory.Move(backup, target); }
                    catch (Exception error) { restoreError = restoreError == null ? error : new AggregateException(restoreError, error); }
                    if (restoreError != null) throw new IOException("Installation failed and the previous plugin could not be fully restored.", restoreError);
                    throw;
                } finally {
                    try { if (Directory.Exists(staging)) Directory.Delete(staging, true); } catch { }
                }
            }
        }

        internal static void Uninstall(string extensionsRoot) {
            var target = TargetPath(extensionsRoot);
            if (Directory.Exists(target)) Directory.Delete(target, true);
        }

        internal static void EnableCepDebugMode() {
            for (int version = 9; version <= 20; version++) {
                using (var key = Registry.CurrentUser.CreateSubKey(@"Software\Adobe\CSXS." + version)) {
                    key.SetValue("PlayerDebugMode", "1", RegistryValueKind.String);
                }
            }
        }

        private static MemoryStream CopyToMemory(Stream input) {
            var copy = new MemoryStream();
            input.CopyTo(copy);
            copy.Position = 0;
            return copy;
        }

        private static void ExtractSafely(Stream payload, string staging) {
            Directory.CreateDirectory(staging);
            var prefix = Path.GetFullPath(staging).TrimEnd(Path.DirectorySeparatorChar) + Path.DirectorySeparatorChar;
            using (var zip = new ZipArchive(payload, ZipArchiveMode.Read, true)) {
                foreach (var entry in zip.Entries) {
                    var destination = Path.GetFullPath(Path.Combine(staging, entry.FullName.Replace('/', Path.DirectorySeparatorChar)));
                    if (!destination.StartsWith(prefix, StringComparison.OrdinalIgnoreCase)) throw new InvalidDataException("Unsafe payload path: " + entry.FullName);
                    if (String.IsNullOrEmpty(entry.Name)) { Directory.CreateDirectory(destination); continue; }
                    Directory.CreateDirectory(Path.GetDirectoryName(destination));
                    using (var input = entry.Open()) using (var output = File.Create(destination)) input.CopyTo(output);
                }
            }
        }

        private static void ValidateExtracted(string root) {
            if (!File.Exists(Path.Combine(root, "CSXS", "manifest.xml")) || !File.Exists(Path.Combine(root, "host", "main.jsx"))) {
                throw new InvalidDataException("Extracted plugin is incomplete.");
            }
        }
    }
}
```
